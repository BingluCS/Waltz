cmake_minimum_required(VERSION 3.21)
project(WaltzCPU LANGUAGES C CXX)
option(WALTZ_CPU_BUILD_TESTS "Build portable CPU tests" ON)
option(WALTZ_CPU_SANITIZE "Enable address/undefined sanitizers" OFF)
option(WALTZ_CPU_NOHALO "Whole-line CPU lifting without halo tiles" OFF)
option(WALTZ_CPU_BUILD_GPU_INTEROP "Build CPU/CUDA interoperability tests" OFF)
option(WALTZ_PYTHON_PACKAGE "Install the shared CPU library inside the Python wheel" OFF)
option(WALTZ_BUILD_HDF5_FILTER "Build the optional HDF5 chunk filter (requires HDF5 headers)" OFF)
add_library(waltz_cpu src/codec.cpp src/transform.cpp src/wzp_cpu.cpp src/quantize_avx2.cpp src/morton_decode_avx2.cpp src/repair_avx2.cpp)
target_compile_definitions(waltz_cpu PRIVATE WALTZ_CPU_NOHALO=$<BOOL:${WALTZ_CPU_NOHALO}>)
add_executable(waltz_cpu_cli tools/waltz_cpu_cli.cpp)
target_link_libraries(waltz_cpu_cli PRIVATE waltz_cpu)
add_executable(waltz_cpu_bench tests/bench_cpu.cpp)
target_link_libraries(waltz_cpu_bench PRIVATE waltz_cpu)
target_include_directories(waltz_cpu PUBLIC include PRIVATE src)
target_compile_features(waltz_cpu PUBLIC cxx_std_17)
if(WALTZ_PYTHON_PACKAGE)
  if(NOT BUILD_SHARED_LIBS)
    message(FATAL_ERROR "The Python bindings require BUILD_SHARED_LIBS=ON")
  endif()
  add_custom_target(waltz_python_runtime DEPENDS waltz_cpu)
  install(TARGETS waltz_cpu LIBRARY DESTINATION pywaltz)
endif()
if(WALTZ_BUILD_HDF5_FILTER)
  find_package(HDF5 REQUIRED COMPONENTS C)
  set_target_properties(waltz_cpu PROPERTIES POSITION_INDEPENDENT_CODE ON)
  add_library(h5waltz MODULE ../python/hdf5/filter.cpp)
  target_include_directories(h5waltz PRIVATE ${HDF5_INCLUDE_DIRS})
  # Only the H5Z callback ABI is used. Do not bring a second HDF5 runtime
  # into a process that already loaded one (notably pip wheels of h5py).
  target_link_libraries(h5waltz PRIVATE waltz_cpu)
  if(WALTZ_PYTHON_PACKAGE)
    set_target_properties(h5waltz PROPERTIES INSTALL_RPATH "$ORIGIN/..")
    add_dependencies(waltz_python_runtime h5waltz)
    install(TARGETS h5waltz LIBRARY DESTINATION pywaltz/plugins)
  endif()
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(waltz_cpu PRIVATE -O3 -ffp-contract=off -fno-fast-math -frounding-math)
  if(WALTZ_CPU_SANITIZE)
    target_compile_options(waltz_cpu PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(waltz_cpu PUBLIC -fsanitize=address,undefined)
  endif()
endif()
if(WALTZ_CPU_BUILD_GPU_INTEROP)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  get_filename_component(WALTZ_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
  set(WALTZ_GPU_LIBRARY "${WALTZ_ROOT}/build/libwaltz.so" CACHE FILEPATH "Existing GPU library; never rebuilt here")
  if(NOT EXISTS "${WALTZ_GPU_LIBRARY}")
    message(FATAL_ERROR "WALTZ_GPU_LIBRARY must identify an existing GPU library")
  endif()
  add_library(waltz_existing_gpu SHARED IMPORTED)
  set_target_properties(waltz_existing_gpu PROPERTIES IMPORTED_LOCATION "${WALTZ_GPU_LIBRARY}")
  add_executable(waltz_cpu_gpu_interop tests/interop_gpu.cu)
  target_include_directories(waltz_cpu_gpu_interop PRIVATE "${WALTZ_ROOT}/include")
  target_compile_options(waltz_cpu_gpu_interop PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr>)
  set_target_properties(waltz_cpu_gpu_interop PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_link_libraries(waltz_cpu_gpu_interop PRIVATE waltz_cpu waltz_existing_gpu CUDA::cudart)
  # GPU tests are explicitly invoked by the user after checking device availability.
endif()
if(WALTZ_CPU_BUILD_TESTS)
  enable_testing()
  add_executable(waltz_cpu_test tests/test_codec.cpp)
  target_link_libraries(waltz_cpu_test PRIVATE waltz_cpu)
  add_test(NAME cpu_codec COMMAND waltz_cpu_test)
  foreach(component IN ITEMS quantize morton_decode repair fp_environment transform wzp)
    set(target waltz_cpu_${component}_test)
    add_executable(${target} tests/test_${component}.cpp)
    target_include_directories(${target} PRIVATE src)
    target_link_libraries(${target} PRIVATE waltz_cpu)
    add_test(NAME cpu_${component} COMMAND ${target})
  endforeach()
  target_compile_options(waltz_cpu_transform_test PRIVATE -ffp-contract=off -fno-fast-math)
endif()
